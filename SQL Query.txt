/*WITH RoomSchedSummary AS ( */
/* Summarizing each room by OR blocks */
SELECT
CONVERT(DATE, START_DTTM) as SCHEDULE_DATE,
slot.ROOM_ID,
START_DTTM,
END_DTTM,
CASE
WHEN CONVERT(TIME, START_DTTM) < '12:00:00' THEN 'AM'
ELSE 'PM'
END AS SESSION_PERIOD,
SLOT_LENGTH,
utl.BLOCK_ID,
CASE
WHEN blk.SURGICAL_SERVICE_C IS NOT NULL THEN ZC_OR_SERVICE.NAME
WHEN blk.PROVIDER_ID IS NOT NULL THEN SURG.PROV_NAME
WHEN blk.SURGEON_GROUP_ID IS NOT NULL THEN OR_GRP.GROUP_NAME
END AS BLOCK_NAME,
MODIFIER_BLOCK_ID,
RESP.PROV_NAME AS RESPONSIBLE_SURGEON,
row_number() over (partition by CONVERT(DATE,START_DTTM),slot.ROOM_ID order by START_DTTM) as ROW_NUM

Into #RoomSchedSummary2

FROM UTL_OR_AVAIL_SLOT slot
LEFT JOIN UTILIZATION utl ON utl.UTILIZATION_ID = slot.UTILIZATION_ID
LEFT JOIN BLOCK modBlk ON slot.MODIFIER_BLOCK_ID = modBlk.BLOCK_ID
LEFT JOIN BLOCK blk ON utl.BLOCK_ID=blk.BLOCK_ID
LEFT JOIN ZC_OR_SERVICE ON blk.SURGICAL_SERVICE_C = ZC_OR_SERVICE.SERVICE_C
LEFT JOIN OR_GRP ON blk.SURGEON_GROUP_ID = OR_GRP.GROUP_ID
LEFT JOIN CLARITY_SER SURG ON blk.PROVIDER_ID = SURG.PROV_ID
LEFT JOIN CLARITY_SER RESP ON modBlk.RESPONSIBLE_PROV_ID = RESP.PROV_ID
WHERE
UTILIZATION_TYPE_C = '1' /*only show block util */
AND blk.BLOCK_TYPE_C <> '0' /* don't show unblocked time */
GROUP BY slot.ROOM_ID, START_DTTM, END_DTTM, utl.BLOCK_ID, MODIFIER_BLOCK_ID, slot.SLOT_LENGTH, blk.SURGICAL_SERVICE_C, NAME, blk.PROVIDER_ID, SURG.PROV_NAME, RESP.PROV_NAME, blk.SURGEON_GROUP_ID, GROUP_NAME
/*), */
;
/*Get all Scheduled Encounters */
/*SchedEncounters AS ( */
SELECT
pat.PAT_ENC_CSN_ID,
pat.CONTACT_DATE AS SCHED_DATE,
CASE
WHEN cs.OR_ID IS NOT NULL THEN cs.OR_ID
WHEN appt.PROV_ID IS NOT NULL THEN appt.PROV_ID
ELSE 'Err'
END AS ROOM_ID,
CASE
WHEN cs.CASE_BEGIN_INSTANT IS NOT NULL THEN DATEADD(MINUTE,cs.SETUP_OFFSET,cs.CASE_BEGIN_INSTANT) /*Add setup time to case start time to get scheduled op time start */
WHEN appt.APPT_DTTM IS NOT NULL THEN appt.APPT_DTTM
ELSE '2000-01-01 00:00:00.000'
END AS SCHED_START,
CASE
WHEN cs.CASE_END_INSTANT IS NOT NULL THEN DATEADD(MINUTE,-cs.CLEANUP_OFFSET,cs.CASE_END_INSTANT) /*Add negative cleanup time to end time to get scheduled op time end */
WHEN appt.APPT_LENGTH IS NOT NULL THEN DATEADD(MINUTE,appt.APPT_LENGTH,appt.APPT_DTTM)
ELSE '2000-01-01 00:00:00.000'
END AS SCHED_END,
pl.LOG_ID,
stTime.TIMING_EVENT_DTTM AS PERF_START,
endTime.TIMING_EVENT_DTTM AS PERF_END,
adm.ADT_PAT_CLASS_C,
cs.PAT_CLASS_C,
cs.SERVICE_C AS SPECIALTY,
reg.REGADDL_ENC_1_C AS FUNDING_STREAM

Into #SchedEncounters2

FROM V_PAT_ENC pat
LEFT JOIN V_SCHED_APPT appt ON appt.PAT_ENC_CSN_ID = pat.PAT_ENC_CSN_ID
LEFT JOIN OR_SER_SURG_RECTYP orSER ON appt.PROV_ID = orSer.PROV_ID
LEFT JOIN PAT_OR_ADM_LINK pl ON pat.PAT_ENC_CSN_ID = pl.PAT_ENC_CSN_ID
LEFT JOIN PAT_ENC_HSP adm ON pl.OR_LINK_CSN = adm.PAT_ENC_CSN_ID
LEFT JOIN REGADDL_ENC reg ON adm.PAT_ENC_CSN_ID = reg.PAT_ENC_CSN_ID
LEFT JOIN OR_CASE cs ON pl.CASE_ID = cs.OR_CASE_ID
LEFT JOIN OR_LOG_TIMING_EVENTS stTime ON
pl.LOG_ID = stTime.LOG_ID AND stTime.TIMING_EVENT_C = '10010'
LEFT JOIN OR_LOG_TIMING_EVENTS endTime ON
pl.LOG_ID = endTime.LOG_ID AND endTime.TIMING_EVENT_C = '10020'
WHERE
(
(pat.ENC_TYPE_C IN ('50','3') AND appt.APPT_DTTM IS NOT NULL AND appt.PROV_ID IS NOT NULL AND orSER.REC_TYPE_C = '3' AND appt.APPT_STATUS_C <> '3')
OR
(pat.ENC_TYPE_C = '51' AND cs.OR_ID IS NOT NULL)
)
/*), */
;
/*Get Scheduled encounters in session */
/*ScheduledInSession AS ( */
SELECT
sched.SCHED_DATE,
sched.ROOM_ID,
sched.PAT_ENC_CSN_ID,
sched.LOG_ID,
sched.SCHED_START,
sched.SCHED_END,
DATEDIFF(MINUTE, sched.SCHED_START, sched.SCHED_END) AS SCHED_OPERATION_TIME,
sched.PERF_START,
sched.PERF_END,
DATEDIFF(MINUTE, sched.PERF_START,sched.PERF_END) AS PERF_OPERATION_TIME,
CASE WHEN (sched.PAT_CLASS_C = '103' OR sched.ADT_PAT_CLASS_C = '103') THEN 1 ELSE 0 END AS EMERGENCY,
rs.BLOCK_ID,
rs.MODIFIER_BLOCK_ID,
rs.START_DTTM AS SESSION_START,
rs.END_DTTM AS SESSION_END,
rs.SESSION_PERIOD,
rs2.END_DTTM AS PREV_BLOCK_END,
sched.FUNDING_STREAM,
sched.SPECIALTY

Into #ScheduledInSession2

FROM #SchedEncounters2 sched
LEFT JOIN #RoomSchedSummary2 rs ON sched.ROOM_ID = rs.ROOM_ID AND sched.SCHED_DATE = rs.SCHEDULE_DATE
LEFT JOIN #RoomSchedSummary2 rs2 ON rs.ROOM_ID = rs2.ROOM_ID AND rs.SCHEDULE_DATE = rs2.SCHEDULE_DATE
AND rs.ROW_NUM = (rs2.ROW_NUM + 1)
WHERE
(
(rs.START_DTTM <= SCHED_START AND sched.SCHED_START <= rs.END_DTTM) /*scheduled start between block start and block end */
OR
(rs.SESSION_PERIOD = 'AM' AND rs.START_DTTM < sched.SCHED_END AND sched.SCHED_START <= rs.END_DTTM) /*scheduled start before block end and ended after block start, AM sessions */
OR
(rs.SESSION_PERIOD = 'PM' AND sched.SCHED_END >= rs.START_DTTM AND sched.SCHED_START <= rs.END_DTTM AND rs2.END_DTTM IS NULL) /*scheduled start before block end and ended after block start, PM sessions w/ no AM */
OR
(rs.SESSION_PERIOD = 'PM' AND sched.SCHED_START > rs2.END_DTTM AND sched.SCHED_END >= rs.START_DTTM AND sched.SCHED_END <= rs.END_DTTM) /*scheduled start after previous block end and ended during block, PM sessions */
)
/*), */
;
/*Start calculations for removing overlapping operation minutes from performed encounters*/

/* STEP 1: Expand each performed row into 1-minute increments */
With MinuteExpanded AS (
SELECT
LOG_ID,
ROOM_ID,
SCHED_DATE,
BLOCK_ID,
DATEADD(MINUTE, v.number, DATEADD(MINUTE, DATEDIFF(MINUTE, 0, PERF_START), 0)) AS MinuteStamp, /*Round Minute Stamp to the nearest minute so we don't double-count minutes with miliseconds defined */
PERF_START
FROM #ScheduledInSession2
JOIN master.dbo.spt_values v ON v.type = 'P' AND v.number < DATEDIFF(MINUTE, PERF_START, PERF_END)
),
/* STEP 2: Assign each minute to the earliest LOG_ID that claimed it */
RankedMinutes AS (
SELECT
LOG_ID,
ROOM_ID,
SCHED_DATE,
BLOCK_ID,
MinuteStamp,
ROW_NUMBER() OVER (
PARTITION BY ROOM_ID, SCHED_DATE, BLOCK_ID, MinuteStamp
ORDER BY PERF_START
) AS minute_rank
FROM MinuteExpanded
),

/* STEP 3: Keep only the first LOG_ID per minute (deduplication) */
DedupedMinutes AS (
SELECT
LOG_ID,
ROOM_ID,
SCHED_DATE,
BLOCK_ID,
COUNT(*) AS DEDUP_PERF_OPERATION_TIME
FROM RankedMinutes
WHERE minute_rank = 1
GROUP BY LOG_ID, ROOM_ID, SCHED_DATE, BLOCK_ID
)/*, */

/*OperationTimesPerSession AS ( */
SELECT
rs.ROOM_ID,
rs.SCHEDULE_DATE,
rs.BLOCK_ID,
rs.MODIFIER_BLOCK_ID,
rs.SESSION_PERIOD,
SUM(s.SCHED_OPERATION_TIME) AS TOTAL_OPERATION_TIME_SCHED,
SUM(s.PERF_OPERATION_TIME) AS TOTAL_OPERATION_TIME_PERF,
SUM(ISNULL(d.DEDUP_PERF_OPERATION_TIME, 0)) AS NONOVERLAP_PERF_OPERATION_TIME,
SUM((s.PERF_OPERATION_TIME - ISNULL(d.DEDUP_PERF_OPERATION_TIME, 0))) AS OVERLAP_MINS,
COUNT(CASE WHEN s.LOG_ID IS NOT NULL AND (s.PERF_START IS NULL OR s.PERF_END IS NULL) THEN 1 END) AS NUM_EXCLUSIONS_OT, /*exclusions (number of logs without either a start or end*/
MIN(s.PERF_START) AS FIRST_START_TIME_PERF,
MAX(s.PERF_END) AS LAST_END_TIME_PERF,
DATEDIFF(MINUTE, MIN(s.PERF_START), MAX(s.PERF_END)) AS RUN_TIME_MINUTES_PERF,
MIN(s.SCHED_START) AS FIRST_START_TIME_SCHED,
MAX(s.SCHED_END) AS LAST_END_TIME_SCHED,
DATEDIFF(MINUTE, MIN(s.SCHED_START), MAX(s.SCHED_END)) AS RUN_TIME_MINUTES_SCHED,
s.PREV_BLOCK_END,
COUNT(DISTINCT s.PAT_ENC_CSN_ID) AS NUM_ENCOUNTERS,
SUM(s.EMERGENCY) AS NUM_EMERGENCY,
/*Early/Late Starts (10 min threshold) */
CASE
WHEN DATEDIFF(MINUTE, rs.START_DTTM, MIN(s.PERF_START)) > 10
THEN 'Late Start'
WHEN DATEDIFF(MINUTE, MIN(s.PERF_START), rs.START_DTTM) > 10
THEN 'Early Start'
ELSE 'On Time'
END AS EARLY_LATE_START,
CASE
WHEN DATEDIFF(MINUTE, rs.START_DTTM, MIN(s.PERF_START)) > 10
THEN DATEDIFF(MINUTE,rs.START_DTTM, MIN(s.PERF_START))
WHEN DATEDIFF(MINUTE, MIN(s.PERF_START), rs.START_DTTM) > 10
THEN DATEDIFF(MINUTE, MIN(s.PERF_START), rs.START_DTTM)
ELSE NULL
END AS EARLY_LATE_START_MINS,

/* Early/Late Ends (10 min threshold) */
CASE
WHEN DATEDIFF(MINUTE, MAX(s.PERF_END), rs.END_DTTM) > 10
THEN 'Early Finish'
WHEN DATEDIFF(MINUTE, rs.END_DTTM, MAX(s.PERF_END)) > 10
THEN 'Late Finish'
ELSE 'On Time'
END AS EARLY_LATE_END,
CASE
WHEN DATEDIFF(MINUTE, MAX(s.PERF_END), rs.END_DTTM) > 10
THEN DATEDIFF(MINUTE, MAX(s.PERF_END), rs.END_DTTM)
WHEN DATEDIFF(MINUTE, rs.END_DTTM, MAX(s.PERF_END)) > 10
THEN DATEDIFF(MINUTE, rs.END_DTTM, MAX(s.PERF_END))
ELSE NULL
END AS EARLY_LATE_END_MINS

Into #OperationTimesPerSession2

FROM #RoomSchedSummary2 rs
LEFT JOIN #ScheduledInSession2 s ON rs.ROOM_ID=s.ROOM_ID
AND rs.BLOCK_ID = s.BLOCK_ID
AND (rs.MODIFIER_BLOCK_ID = s.MODIFIER_BLOCK_ID OR rs.MODIFIER_BLOCK_ID IS NULL)
AND rs.SCHEDULE_DATE = s.SCHED_DATE
AND rs.SESSION_PERIOD = s.SESSION_PERIOD
LEFT JOIN DedupedMinutes d ON s.LOG_ID = d.LOG_ID
WHERE rs.SCHEDULE_DATE >= (CONVERT(DATETIME, '2026-10-24', 120))

GROUP BY rs.SCHEDULE_DATE, rs.ROOM_ID, rs.BLOCK_ID, rs.MODIFIER_BLOCK_ID, rs.SESSION_PERIOD, s.PREV_BLOCK_END, rs.START_DTTM, rs.END_DTTM
/*) */
;

SELECT GROUPING_ID( orLoc.SERV_AREA_ID, orSER.LOC_ID, rs.BLOCK_ID, rs.ROOM_ID) AS "LEVEL_BIT_VECTOR",
orLoc.SERV_AREA_ID AS "SUMMARY_TARGET_1",
orSer.LOC_ID AS "SUMMARY_TARGET_2",
rs.BLOCK_ID AS "SUMMARY_TARGET_3",
rs.ROOM_ID AS "SUMMARY_TARGET_4",
rs.SCHEDULE_DATE,
1,
SUM(ot.NONOVERLAP_PERF_OPERATION_TIME) AS NOVERLAP_OP_TIME_MINS_PERF, /*total performed operation time in minutes for a given block */
SUM(rs.SLOT_LENGTH) AS OP_SESSION_LENGTH_MINUTES, /* Session Length in Minutes */
SUM(ot.NUM_EXCLUSIONS_OT) AS NUM_EXCLUSIONS_OT, /* Exclusions (missing either start time or end time) */
2,
SUM(ot.RUN_TIME_MINUTES_PERF) AS RUN_TIME_MINS_PERF, /* total performed run time minutes for a given block */
SUM(rs.SLOT_LENGTH) AS RUN_SESSION_LENGTH_MINUTES, /*Session Length in Minutes */
COUNT(CASE WHEN ot.RUN_TIME_MINUTES_PERF IS NULL THEN 1 END) AS NUM_EXCLUSIONS_RT, /* number of sessions where run time couldn't be calculated */
3,
SUM(CASE WHEN ot.EARLY_LATE_START = 'Early Start' THEN ot.EARLY_LATE_START_MINS END) AS EARLY_START_MINUTES, /* total minutes from early starts */
4,
SUM(CASE WHEN ot.EARLY_LATE_START = 'Late Start' THEN ot.EARLY_LATE_START_MINS END) AS LATE_START_MINUTES, /* total minutes from late starts */
5,
SUM(CASE WHEN ot.EARLY_LATE_END = 'Early Finish' THEN ot.EARLY_LATE_END_MINS END) AS EARLY_FINISH_MINUTES, /* total minutes from early finishes */
6,
SUM(CASE WHEN ot.EARLY_LATE_END = 'Late Finish' THEN ot.EARLY_LATE_END_MINS END) AS LATE_FINISH_MINUTES, /* total minutes from late finishes */
7,
SUM((ot.NUM_ENCOUNTERS - ot.NUM_EXCLUSIONS_OT)) AS NUM_ENCOUNTERS_PERF, /*number of performed encounters per session (null LOG_ID not counted with DISTINCT for sessions with no encounters) */
8,
SUM(ot.NUM_EMERGENCY) AS NUM_EMERGENCY_PERF, /*number of emergency patients (based on admission's ADT patient class) in a session */
9,
SUM(ot.TOTAL_OPERATION_TIME_SCHED) AS OP_TIME_MINS_SCHED, /*total scheduled operation time in minutes for a given block */
SUM(rs.SLOT_LENGTH) AS OP_SESSION_LENGTH_MINUTES, /* Session Length in Minutes */
SUM(0) AS NUM_EXCLUSIONS_SCHED_OT,
10,
SUM(ot.RUN_TIME_MINUTES_SCHED) AS RUN_TIME_MINS_SCHED, /* total scheduled run time minutes for a given block */
SUM(rs.SLOT_LENGTH) AS RUN_SESSION_LENGTH_MINUTES, /*Session Length in Minutes */
SUM(0) AS NUM_EXCLUSIONS_SCHED_RT,
11,
SUM(ot.NUM_ENCOUNTERS) AS NUM_ENCOUNTERS_SCHED, /*number of scheduled encounters per session */
12,
SUM(ot.NUM_EMERGENCY) AS NUM_EMERGENCY_SCHED, /*number of emergency patients (based on admission's ADT patient class) in a session */
13,
COUNT(rs.BLOCK_ID) AS NUM_SESSIONS,
14,
SUM(rs.SLOT_LENGTH) AS SESSION_MINUTES
FROM #RoomSchedSummary2 rs
LEFT JOIN OR_SER_SURG_SRVC orSer ON rs.ROOM_ID = orSer.PROV_ID
LEFT JOIN CLARITY_LOC orLoc ON orSer.LOC_ID = orLoc.LOC_ID
LEFT JOIN #OperationTimesPerSession2 ot ON
rs.ROOM_ID = ot.ROOM_ID
AND rs.SCHEDULE_DATE = ot.SCHEDULE_DATE
AND rs.BLOCK_ID = ot.BLOCK_ID
AND rs.SESSION_PERIOD = ot.SESSION_PERIOD
AND ((rs.MODIFIER_BLOCK_ID = ot.MODIFIER_BLOCK_ID) or rs.MODIFIER_BLOCK_ID is Null)
WHERE
rs.SCHEDULE_DATE >= (CONVERT(DATETIME, '2026-10-24', 120))
AND rs.BLOCK_ID IS NOT NULL
/*GROUP BY GROUPING SETS (orLoc.SERV_AREA_ID, orSER.LOC_ID, rs.BLOCK_ID, rs.ROOM_ID), rs.SCHEDULE_DATE */
GROUP BY GROUPING SETS
(
orLoc.SERV_AREA_ID /*Service Area*/
, (orLoc.SERV_AREA_ID, orSER.LOC_ID) /*Service Area ^ OR/INV Location*/
, (orLoc.SERV_AREA_ID, rs.BLOCK_ID) /*Service Area ^ Block*/
, (orLoc.SERV_AREA_ID, rs.ROOM_ID) /*Service Area ^ ROOM_ID*/
)
, rs.SCHEDULE_DATE

procs.OTHER_PROCS

into #ScheduledInSession

FROM #SchedEncounters sched
LEFT JOIN #RoomSchedSummary rs ON
sched.ROOM_ID = rs.ROOM_ID
AND sched.SCHED_DATE = rs.SCHEDULE_DATE
LEFT JOIN #RoomSchedSummary rs2 ON
rs.ROOM_ID = rs2.ROOM_ID
AND rs.SCHEDULE_DATE = rs2.SCHEDULE_DATE
AND rs.ROW_NUM = (rs2.ROW_NUM + 1)

LEFT OUTER JOIN #procs procs ON Sched.PAT_ENC_CSN_ID = Procs.PAT_ENC_CSN_ID

WHERE
(
(rs.START_DTTM <= SCHED_START AND sched.SCHED_START <= rs.END_DTTM) /*scheduled start between block start and block end */
OR
(rs.SESSION_PERIOD = 'AM' AND rs.START_DTTM < sched.SCHED_END AND sched.SCHED_START <= rs.END_DTTM) /*scheduled start before block end and ended after block start, AM sessions */
OR
(rs.SESSION_PERIOD = 'PM' AND sched.SCHED_END >= rs.START_DTTM AND sched.SCHED_START <= rs.END_DTTM AND rs2.END_DTTM IS NULL) /*scheduled start before block end and ended after block start, PM sessions w/ no AM */
OR
(rs.SESSION_PERIOD = 'PM' AND sched.SCHED_START > rs2.END_DTTM AND sched.SCHED_END >= rs.START_DTTM AND sched.SCHED_END <= rs.END_DTTM) /*scheduled start after previous block end and ended during block, PM sessions */
)
/*), */
;
/*Start calculations for removing overlapping operation minutes from performed encounters*/

/* STEP 1: Expand each performed row into 1-minute increments */
With MinuteExpanded AS (
SELECT
LOG_ID,
ROOM_ID,
SCHED_DATE,
BLOCK_ID,
DATEADD(MINUTE, v.number, DATEADD(MINUTE, DATEDIFF(MINUTE, 0, PERF_START), 0)) AS MinuteStamp, /*Round Minute Stamp to the nearest minute so we don't double-count minutes with miliseconds defined */
PERF_START
FROM #ScheduledInSession
JOIN master.dbo.spt_values v ON v.type = 'P'
AND v.number < DATEDIFF(MINUTE, PERF_START, PERF_END)
),
/* STEP 2: Assign each minute to the earliest LOG_ID that claimed it */
RankedMinutes AS (
SELECT
LOG_ID,
ROOM_ID,
SCHED_DATE,
BLOCK_ID,
MinuteStamp,
ROW_NUMBER() OVER (
PARTITION BY ROOM_ID, SCHED_DATE, BLOCK_ID, MinuteStamp
ORDER BY PERF_START
) AS minute_rank
FROM MinuteExpanded
),

/* STEP 3: Keep only the first LOG_ID per minute (deduplication) */
DedupedMinutes AS (
SELECT
LOG_ID,
ROOM_ID,
SCHED_DATE,
BLOCK_ID,
COUNT(*) AS DEDUP_PERF_OPERATION_TIME
FROM RankedMinutes
WHERE minute_rank = 1
GROUP BY LOG_ID, ROOM_ID, SCHED_DATE, BLOCK_ID
),

DetailCalcs AS (
SELECT
s.LOG_ID,
s.ROOM_ID,
s.SCHED_DATE,
s.PAT_ENC_CSN_ID,
s.PRY_PROC,
s.OTHER_PROCS,
s.SPECIALTY,
s.SESSION_START,
s.SESSION_END,
s.BLOCK_ID,
s.MODIFIER_BLOCK_ID,
s.SESSION_PERIOD,
s.SCHED_START,
s.SCHED_END,
s.SCHED_OPERATION_TIME,
s.PERF_START,
s.PERF_END,
s.PERF_OPERATION_TIME,
ISNULL(d.DEDUP_PERF_OPERATION_TIME, 0) AS NONOVERLAP_PERF_OPERATION_TIME,
(s.PERF_OPERATION_TIME - ISNULL(d.DEDUP_PERF_OPERATION_TIME, 0)) AS OVERLAP_MINS,
s.EMERGENCY,
s.FUNDING_STREAM,
CASE
WHEN (s.PERF_START > s.SESSION_END) OR (s.PERF_END < s.SESSION_START)
THEN 'No'
ELSE 'Yes'
END AS PERFORMED_IN_SESSION
FROM #ScheduledInSession s
LEFT JOIN DedupedMinutes d ON s.LOG_ID = d.LOG_ID
)
