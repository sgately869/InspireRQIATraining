TimelineIntervals =
VAR Planned =
    SELECTCOLUMNS (
        'SessionReview',
        "Timeline Session ID", 'SessionReview'[Source Session ID],
        "Interval ID", "Plan|" & 'SessionReview'[Source Session ID],
        "Band", "Planned session",
        "Interval Start", 'SessionReview'[Slot Start Time],
        "Interval End", 'SessionReview'[Slot End Time]
    )
VAR RecordedRuns =
    SELECTCOLUMNS (
        FILTER (
            'SessionReview',
            NOT ISBLANK ( 'SessionReview'[Run Minutes] )
                && NOT ISBLANK ( 'SessionReview'[First Actual Start] )
                && NOT ISBLANK ( 'SessionReview'[Last Actual End] )
                && 'SessionReview'[Last Actual End] >= 'SessionReview'[First Actual Start]
        ),
        "Timeline Session ID", 'SessionReview'[Source Session ID],
        "Interval ID", "Run|" & 'SessionReview'[Source Session ID],
        "Band", "Actual run",
        "Interval Start", 'SessionReview'[First Actual Start],
        "Interval End", 'SessionReview'[Last Actual End]
    )
VAR Occupied =
    SELECTCOLUMNS (
        FILTER ( 'TimelineChartData', 'TimelineChartData'[Record Type] = "Operation" ),
        "Timeline Session ID", 'TimelineChartData'[Timeline Session ID],
        "Interval ID", 'TimelineChartData'[Interval ID],
        "Band", "Occupied theatre time",
        "Interval Start", 'TimelineChartData'[Interval Start],
        "Interval End", 'TimelineChartData'[Interval End]
    )
RETURN
    UNION ( Planned, RecordedRuns, Occupied )

// TimelineIntervals > New column. Create each of the following three columns separately.
Session Row =
    RELATED ( 'SessionReview'[Hospital] ) & " | "
        & RELATED ( 'SessionReview'[Operating Theatre] ) & " | "
        & FORMAT ( RELATED ( 'SessionReview'[Slot Start Time] ), "HH:mm" ) & "–"
        & FORMAT ( RELATED ( 'SessionReview'[Slot End Time] ), "HH:mm" )
        & "¦" & 'TimelineIntervals'[Timeline Session ID]

Start Minute =
    1440 * (
        'TimelineIntervals'[Interval Start]
            - RELATED ( 'SessionReview'[Procedure Date] )
    )

End Minute =
    1440 * (
        'TimelineIntervals'[Interval End]
            - RELATED ( 'SessionReview'[Procedure Date] )
    )

// TimelineIntervals > New measure. Use as a visual-level filter on Deneb, greater than 0.
Timeline Visible Rows =
    IF (
        HASONEVALUE ( 'Review Calendar'[Date] ),
        COUNTROWS ( 'TimelineIntervals' )
    )

// Review Calendar > New column. Create each of the following three columns separately.
Timeline Month =
    FORMAT ( 'Review Calendar'[Date], "mmm-yy", "en-GB" )

Timeline Month Sort =
    YEAR ( 'Review Calendar'[Date] ) * 100 + MONTH ( 'Review Calendar'[Date] )

Timeline Procedure Date =
    FORMAT ( 'Review Calendar'[Date], "dd/mm/yyyy", "en-GB" )
